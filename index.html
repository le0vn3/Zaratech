import logging from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup from telegram.ext import ( Application, CommandHandler, CallbackQueryHandler, MessageHandler, filters, ContextTypes, ConversationHandler ) # Configuration du logging logging.basicConfig(level=logging.INFO) # États pour la ConversationHandler (Recherche par Image) WAITING_FOR_PHOTO = 1 # 1. Fonctionnalité : Recherche par Photo async def start_photo_search(update: Update, context: ContextTypes.DEFAULT_TYPE): query = update.callback_query await query.answer() await query.edit_message_text( "📸 Envoyez-moi une photo du vêtement que vous cherchez, " "et je trouverai le modèle équivalent chez Zara !" ) return WAITING_FOR_PHOTO async def process_photo(update: Update, context: ContextTypes.DEFAULT_TYPE): photo_file = await update.message.photo[-1].get_file() # Message de confirmation (ici vous intégreriez votre API d'analyse d'image) await update.message.reply_text( "🔍 Analyse de l'image en cours...\n\n" "✨ **Résultat trouvé chez Zara :**\n" "• **Robe Courte Imprimée** (Réf: 3121/044)\n" "• Prix : 39,95€\n" "• Disponibilité : En stock" ) return ConversationHandler.END async def cancel(update: Update, context: ContextTypes.DEFAULT_TYPE): await update.message.reply_text("Recherche annulée.") return ConversationHandler.END # 4. Lancement du bot def main(): # Remplace 'TON_TOKEN_HERE' par ton véritable token BotFather application = Application.builder().token("TON_TOKEN_HERE").build() # Configuration de la recherche par photo (Conversation) photo_handler = ConversationHandler( entry_points=[CallbackQueryHandler(start_photo_search, pattern="^search_photo$")], states={ WAITING_FOR_PHOTO: [MessageHandler(filters.PHOTO, process_photo)] }, fallbacks=[CommandHandler("cancel", cancel)] ) # Ajout des gestionnaires application.add_handler(CommandHandler("start", start)) application.add_handler(photo_handler) application.add_handler(CallbackQueryHandler(button_handler)) # Démarrage application.run_polling() if __name__ == "__main__": main()
